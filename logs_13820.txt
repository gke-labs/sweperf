        with open(filename, 'r') as f:

        print(f"File {filename} not found")

        print(f"End line {end_line} out of range")

    # Adjust for 0-based indexing

        f.writelines(lines)
        
    print("File updated successfully")

if __name__ == "__main__":
    main()
